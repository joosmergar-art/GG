-- Evitar ejecuciones duplicadas
if _G.SpeedSimulatorHubLoaded then return end
_G.SpeedSimulatorHubLoaded = true

if not game:IsLoaded() then game.Loaded:Wait() end

---------------------------------------------------------
-- ⚡ SERVICIOS Y CONFIGURACIÓN DEL REMOTO
---------------------------------------------------------
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local AddSpeedEvent = ReplicatedStorage:WaitForChild("Events"):WaitForChild("AddSpeed")

local Settings = {
    AutoSpeed = false,
    Delay = 0.01 -- Velocidad de spam masivo
}

---------------------------------------------------------
-- 🚀 HILO DE FARMEO ULTRA RÁPIDO
---------------------------------------------------------
task.spawn(function()
    while true do
        if Settings.AutoSpeed then
            -- Enviamos 15 peticiones simultáneas por ciclo para inflar los pasos/velocidad
            for i = 1, 15 do
                task.spawn(function()
                    pcall(function()
                        AddSpeedEvent:FireServer()
                    end)
                end)
            end
            task.wait(Settings.Delay)
        else
            task.wait(0.5)
        end
    end
end)

---------------------------------------------------------
-- 🎨 INTERFAZ GRÁFICA (RAYFIELD)
---------------------------------------------------------
local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()
local Window = Rayfield:CreateWindow({
   Name = "⚡ Speed Simulator Hub ⚡",
   LoadingTitle = "Inyectando Conexión de Velocidad...",
   LoadingSubtitle = "Optimizado por Gemini AI",
   ConfigurationSaving = { Enabled = false },
   KeySystem = false
})

local MainTab = Window:CreateTab("Auto Farm", 4483362458)

MainTab:CreateToggle({
   Name = "👟 Auto Add Speed (Ráfaga)",
   CurrentValue = false,
   Flag = "ToggleAutoSpeed",
   Callback = function(Value) Settings.AutoSpeed = Value end,
})

Rayfield:Notify({
   Name = "¡Inyector Listo!",
   Content = "Sistema de velocidad sincronizado correctamente.",
   Duration = 5
})
