-- Evitar ejecuciones duplicadas
if _G.SpeedSimulatorUnifiedHubLoaded then return end
_G.SpeedSimulatorUnifiedHubLoaded = true

if not game:IsLoaded() then game.Loaded:Wait() end

---------------------------------------------------------
-- ⚡ SERVICIOS Y CONFIGURACIÓN DE REMOTOS
---------------------------------------------------------
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local EventsFolder = ReplicatedStorage:WaitForChild("Events")

-- Remotos del juego de velocidad
local AddSpeedEvent = EventsFolder:WaitForChild("AddSpeed")
local RequestRebirthRF = EventsFolder:WaitForChild("RequestRebirth")

local Settings = {
    AutoSpeed = false,
    AutoRebirth = false,
    SpeedDelay = 0.01,   -- Ráfaga masiva para correr
    RebirthDelay = 1     -- Chequeo seguro para RemoteFunction (1 segundo)
}

---------------------------------------------------------
-- 🚀 HILOS EN PARALELO (MÁXIMO RENDIMIENTO)
---------------------------------------------------------

-- Hilo 1: Auto Add Speed (Multi-Hilo)
task.spawn(function()
    while true do
        if Settings.AutoSpeed then
            for i = 1, 15 do
                task.spawn(function()
                    pcall(function()
                        AddSpeedEvent:FireServer()
                    end)
                end)
            end
            task.wait(Settings.SpeedDelay)
        else
            task.wait(0.5)
        end
    end
end)

-- Hilo 2: Auto Rebirth (Optimizado para InvokeServer)
task.spawn(function()
    while true do
        if Settings.AutoRebirth then
            -- Usamos task.spawn interno para que el InvokeServer no trabe el script principal
            task.spawn(function()
                pcall(function()
                    RequestRebirthRF:InvokeServer()
                end)
            end)
            task.wait(Settings.RebirthDelay)
        else
            task.wait(0.5)
        end
    end
end)

---------------------------------------------------------
-- 🎨 INTERFAZ GRÁFICA (RAYFIELD)
---------------------------------------------------------
local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()
local Window = Rayfield:CreateWindow({
   Name = "⚡ Speed & Rebirth Hub ⚡",
   LoadingTitle = "Unificando Sistemas de Red...",
   LoadingSubtitle = "Optimizado por Gemini AI",
   ConfigurationSaving = { Enabled = false },
   KeySystem = false
})

local MainTab = Window:CreateTab("Auto Farm", 4483362458)

-- === CONTROLES ===

MainTab:CreateToggle({
   Name = "👟 Auto Add Speed (Ráfaga)",
   CurrentValue = false,
   Flag = "ToggleSpeedFarm",
   Callback = function(Value) Settings.AutoSpeed = Value end,
})

MainTab:CreateToggle({
   Name = "♻️ Auto Rebirth (Seguro)",
   CurrentValue = false,
   Flag = "ToggleSpeedRebirth",
   Callback = function(Value) Settings.AutoRebirth = Value end,
})

Rayfield:Notify({
   Name = "¡Combo Listo!",
   Content = "Velocidad y Rebirth vinculados correctamente en Delta.",
   Duration = 5
})
